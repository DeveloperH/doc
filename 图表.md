```html
        // --- 設定與選取元素 ---
        const canvas = document.getElementById('progressCanvas');
        const ctx = canvas.getContext('2d');
        const rangeInput = document.getElementById('percentRange');

        // 核心參數 (對應 400x400 的物理解析度)
        const size = 400; 
        const center = size / 2; // 200
        const radius = 160;     // 圓環半徑
        const lineWidth = 48;   // 圓環粗細
        const startAngle = -Math.PI / 2; // 12點鐘方向 (-90度)

        // 顏色定義 (對應圖片)
        const colorBg = "#2d3442";      // 背景深色圓環
        const colorBar = "#d7943d";     // 前景橘色進度

        const colorLabel = "#717784";   // "得分"文字

            // 1. 清空畫布
            ctx.clearRect(0, 0, size, size);

            ctx.arc(center, center, radius, 0, 2 * Math.PI); // 繪製完整圓形

            ctx.lineWidth = lineWidth;
            ctx.lineCap = 'butt'; // 背景環不需要圓端點
            ctx.stroke();

            // --- B. 繪製橘色進度環 ---
            ctx.beginPath();
            // arc(cx, cy, r, 開始角度, 結束角度)
            ctx.arc(center, center, radius, startAngle, currentAngle); 
            ctx.strokeStyle = colorBar;
            ctx.lineWidth = lineWidth;
            ctx.lineCap = 'round'; // **關鍵：讓進度條兩端均為圓角**
            ctx.stroke();

            // --- C. 繪製置中文字 ---
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';

            // 1. 繪製分數 (60)
            ctx.fillStyle = colorScore;
            ctx.font = 'bold 128px -apple-system, sans-serif'; // 物理尺寸需要加大
            ctx.fillText(value, center, center - 20); // 稍微往上偏

            // 2. 繪製 "得分" 標籤
            ctx.fillStyle = colorLabel;
            ctx.font = '48px sans-serif';
            ctx.fillText('得分', center, center + 70); // 稍微往下偏
        }

        // --- 監聽事件與初始繪製 ---

        // 1. 監聽滑桿數值變化
        rangeInput.addEventListener('input', (e) => {
            drawProgress(e.target.value);
        });

        // 2. 初始繪製 (60分)
        drawProgress(60);

    </script>

</body>
</html>
```

```vue
<template>
  <view class="container">
    <view class="canvas-wrapper">
      <canvas canvas-id="progressCanvas" class="progress-canvas"></canvas>
    </view>

    <view class="controls">
      <text class="label">調整分數測試 (放開滑桿後更新)</text>
      <slider :value="score" min="0" max="100" activeColor="#d7943d" backgroundColor="#2d3442" block-color="#d7943d"
        @change="onSliderChange" class="slider" />
    </view>
  </view>
</template>

<script>
  export default {
    data() {
      return {
        score: 60 // 預設分數 60 分
      };
    },
    mounted() {
      // 延遲 200ms 確保 App 原生畫布組件完全載入就緒，避免首次繪製失敗
      setTimeout(() => {
        this.initCanvas();
      }, 200);
    },
    methods: {
      // 初始化 Canvas
      initCanvas() {
        this.canvasCtx = uni.createCanvasContext('progressCanvas', this);
        // 首次繪製
        this.drawProgress(this.score);
      },

      // 繪製進度條
      drawProgress(value) {
        if (!this.canvasCtx) return;

        const ctx = this.canvasCtx;

        // 🎯 固定 200x200 畫布的邏輯座標
        const center = 100; // 圓心 (200/2)
        const radius = 80; // 半徑
        const lineWidth = 24; // 圓環粗細
        const startAngle = -Math.PI / 2; // 12點鐘方向 (-90度)

        // 1. 清空畫布 (200x200)
        ctx.clearRect(0, 0, 200, 200);

        // 2. 計算結束角度
        const percent = value / 100;
        const currentAngle = startAngle + (percent * (2 * Math.PI));

        // 3. 繪製深色背景圓環
        ctx.beginPath();
        ctx.arc(center, center, radius, 0, 2 * Math.PI);
        ctx.strokeStyle = "#2d3442";
        ctx.lineWidth = lineWidth;
        ctx.lineCap = 'butt';
        ctx.stroke();

        // 4. 繪製橘色進度環
        ctx.beginPath();
        ctx.arc(center, center, radius, startAngle, currentAngle);
        ctx.strokeStyle = "#d7943d";
        ctx.lineWidth = lineWidth;
        ctx.lineCap = 'round'; // 圓角端點
        ctx.stroke();

        // 5. 繪製中央分數文字
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillStyle = "#ffffff";
        ctx.font = 'bold 60px sans-serif';
        ctx.fillText(value.toString(), center, center - 10);

        // 6. 繪製 "得分" 標籤
        ctx.fillStyle = "#717784";
        ctx.font = '20px sans-serif';
        ctx.fillText('得分', center, center + 35);

        // 7. 渲染到畫面上
        ctx.draw(false);
      },

      // 🎯 監聽 slider 的 @change 事件
      onSliderChange(e) {
        this.score = e.detail.value;
        this.drawProgress(this.score);
      }
    }
  };
</script>
```

```vue
  <view class="progress-container" :style="{ width: size + 'px', height: size + 'px' }">

<script>
  export default {
    name: "CircleProgress",
    props: {
      // 唯一的 Canvas ID (若在同一個頁面放多個組件，請務必傳入不同的 ID)
      canvasId: {
        type: String,
        default: "circleProgressCanvas"
      },
      // 目前進度分數 (0 - 100)
      value: {
        type: Number,
        default: 60
      },
      // 圓環的整體大小 (單位 px)
      size: {
        type: Number,
        default: 200
      },
      // 圓環的粗細 (單位 px)
      lineWidth: {
        type: Number,
        default: 24
      },
      // 進度條的主顏色 (橘色)
      activeColor: {
        type: String,
        default: "#d7943d"
      },
      // 背景圓環的顏色 (深藍灰)
      backgroundColor: {
        type: String,
        default: "#2d3442"
      },
      // 中央分數的文字顏色
      textColor: {
        type: String,
        default: "#ffffff"
      },
      // 中央分數的字體大小 (單位 px)
      textSize: {
        type: Number,
        default: 60
      },
      // 標籤文字 (例如：'得分'、'完成度')
      label: {
        type: String,
        default: "得分"
      },
      // 標籤文字顏色
      labelColor: {
        type: String,
        default: "#717784"
      },
      // 標籤文字大小 (單位 px)
      labelSize: {
        type: Number,
        default: 20
      }
    },
    data() {
      return {
        canvasCtx: null
      };
    },
    watch: {
      // 🎯 核心：當父組件傳進來的值改變時，觸發重新繪製
      value(newVal) {
        this.drawProgress(newVal);
      }
    },
    mounted() {
      // 延遲 200ms 確保 App 端原生畫布組件就緒
      setTimeout(() => {
        this.initCanvas();
      }, 200);
    },
    methods: {
      // 初始化組件專屬的 Canvas 上下文
      initCanvas() {
        // 在組件內獲取上下文，必須傳入 this 確保作用域正確
        this.canvasCtx = uni.createCanvasContext(this.canvasId, this);
        this.drawProgress(this.value);
      },

      // 執行繪製
      drawProgress(value) {
        if (!this.canvasCtx) return;

        const ctx = this.canvasCtx;
        const size = this.size;

        // 動態依據傳入的 size 計算圓心與半徑
        const center = size / 2;
        const radius = (size - this.lineWidth) / 2;
        const startAngle = -Math.PI / 2; // 12點鐘方向

        // 1. 清空畫布
        ctx.clearRect(0, 0, size, size);

        // 2. 計算結束角度 (限制在 0-100 之間)
        const safeValue = Math.min(Math.max(value, 0), 100);
        const percent = safeValue / 100;
        const currentAngle = startAngle + (percent * (2 * Math.PI));

        // 3. 繪製背景底環
        ctx.beginPath();
        ctx.arc(center, center, radius, 0, 2 * Math.PI);
        ctx.strokeStyle = this.backgroundColor;
        ctx.lineWidth = this.lineWidth;
        ctx.lineCap = "butt";
        ctx.stroke();

        // 4. 繪製進度環
        ctx.beginPath();
        ctx.arc(center, center, radius, startAngle, currentAngle);
        ctx.strokeStyle = this.activeColor;
        ctx.lineWidth = this.lineWidth;
        ctx.lineCap = "round"; // 完美滑順圓角
        ctx.stroke();

        // 5. 繪製中央主要分數 (等比例微調高度)
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";
        ctx.fillStyle = this.textColor;
        ctx.font = `bold ${this.textSize}px sans-serif`;
        ctx.fillText(safeValue.toString(), center, center - (size * 0.05));

        // 6. 繪製下方標籤 (等比例微調高度)
        ctx.fillStyle = this.labelColor;
        ctx.font = `${this.labelSize}px sans-serif`;
        ctx.fillText(this.label, center, center + (size * 0.175));

        // 7. 渲染生效
        ctx.draw(false);
      }
    }
  };
</script>
```

```vue
<template>
  <view class="progress-container" :style="{ width: ctxSize + 'px', height: ctxSize + 'px' }">
    <canvas :canvas-id="canvasId" :style="{ width: ctxSize + 'px', height: ctxSize + 'px' }"
      class="progress-canvas"></canvas>
  </view>
</template>

<script>
  export default {
    name: "CircleProgress",
    props: {
      canvasId: {
        type: String,
        default: "circleProgressCanvas"
      },
      value: {
        type: Number,
        default: 60
      },
      // 傳入的尺寸全部以 rpx 為單位
      size: {
        type: Number,
        default: 400
      },
      lineWidth: {
        type: Number,
        default: 48
      },
      textSize: {
        type: Number,
        default: 120
      },
      labelSize: {
        type: Number,
        default: 40
      },
      activeColor: {
        type: String,
        default: "#d7943d"
      },
      backgroundColor: {
        type: String,
        default: "#2d3442"
      },
      textColor: {
        type: String,
        default: "#ffffff"
      },
      label: {
        type: String,
        default: "得分"
      },
      labelColor: {
        type: String,
        default: "#717784"
      }
    },
    data() {
      return {
        // 在初始化時就同步算好當前手機對應的 px 寬高
        ctxSize: uni.upx2px(this.size),
        canvasCtx: null,
        isCanvasReady: false // 標記原生畫布是否佈局完成
      };
    },
    watch: {
      // 監聽分數變化
      value(newVal) {
        // 只有當畫布完全準備好時才允許重繪，避免在佈局未完成時強行寫入
        if (this.isCanvasReady) {
          this.drawProgress(newVal);
        }
      }
    },
    mounted() {
      this.$nextTick(() => {
        // 🎯 核心黑魔法：利用 SelectorQuery 確保 App 端原生節點佈局渲染 pass 徹底完成
        uni.createSelectorQuery()
          .in(this)
          .select('.progress-canvas')
          .boundingClientRect(() => {
            this.initCanvas();
          })
          .exec();
      });
    },
    methods: {
      initCanvas() {
        this.canvasCtx = uni.createCanvasContext(this.canvasId, this);
        this.isCanvasReady = true;
        // 第一次安全繪製
        this.drawProgress(this.value);
      },

      drawProgress(value) {
        if (!this.canvasCtx) return;

        const ctx = this.canvasCtx;

        // 將所有 rpx 轉為當前手機最精確的 px 繪製座標
        const sizePx = uni.upx2px(this.size);
        const lineWidthPx = uni.upx2px(this.lineWidth);
        const textSizePx = uni.upx2px(this.textSize);
        const labelSizePx = uni.upx2px(this.labelSize);

        const center = sizePx / 2;
        const radius = (sizePx - lineWidthPx) / 2;
        const startAngle = -Math.PI / 2;

        // 1. 清空畫布
        ctx.clearRect(0, 0, sizePx, sizePx);

        // 2. 計算角度
        const safeValue = Math.min(Math.max(value, 0), 100);
```

```vue
    // 图表宽度 - 支持字符串和数字
    width: {
      type: [Number, String],
      default: '100%'
    },
    // 图表高度 - 支持字符串和数字
    height: {
      type: [Number, String],
      default: '100%'
    },
    // 图表数据
    data: {
      type: Array,
      default: () => []
    },
    // X轴标签
    labels: {
      type: Array,
      default: () => []
    },
    // 【新加】Y轴最小值
    yMin: {
      type: Number,
      default: 0
    },
    // 【新加】Y轴最大值
    yMax: {
      type: Number,
      default: 200
    },
    // 【新加】Y轴刻度间隔
    yInterval: {
      type: Number,
      default: 40
    },
    // 线条颜色
    lineColor: {
      type: String,
      default: '#8B5CF6'
    },
    // 渐变填充颜色
    gradientColors: {
      type: Array,
      default: () => ['rgba(139, 92, 246, 0.4)', 'rgba(139, 92, 246, 0.05)']
    },
    // 数据点颜色
    pointColor: {
      type: String,
      default: '#8B5CF6'
    },
    // 数据点边框颜色
    pointBorderColor: {
      type: String,
      default: '#8B5CF6'
    },
    // 数据点半径
    pointRadius: {
      type: Number,
      default: 6
    },
    // 坐标轴线颜色
    axisColor: {
      type: String,
      default: 'rgba(255, 255, 255, 0.2)'
    },
    // 网格线颜色
    gridColor: {
      type: String,
      default: 'rgba(255, 255, 255, 0.1)'
    },
    // 文字颜色
    textColor: {
      type: String,
      default: 'rgba(255, 255, 255, 0.6)'
    },
    // 背景颜色
    backgroundColor: {
      type: String,
      default: 'transparent'
    }
  },
  data() {
    return {
      chart: null,
      resizeTimer: null // 防抖定时器变量
    };
  },
  computed: {
    chartWrapperStyle() {
      return {
        width: '100%',
        height: '100%',
        backgroundColor: this.backgroundColor,
        borderRadius: '8rpx',
        overflow: 'hidden',
        position: 'relative'
      };
    },
    // 合并数据和标签监听，防止二次渲染闪烁
    chartDataLines() {
      return {
        labels: this.labels,
        data: this.data
      };
    },
    // 【新加】合并 Y 轴配置监听，防止动态改变范围时漏更
    chartYAxis() {
      return {
        min: this.yMin,
        max: this.yMax,
        interval: this.yInterval
      };
    },
    // 合并尺寸监听
    chartSize() {
      return {
        width: this.width,
        height: this.height
      };
    }
  },
  watch: {
    chartDataLines: {
      deep: true,
      handler() {
        this.updateChart();
      }
    },
    // 【新加】监听 Y 轴属性变化
    chartYAxis: {
      deep: true,
      handler() {
        this.updateChart();
      }
    },
    chartSize: {
      deep: true,
      handler() {
        this.handleResize();
      }
    }
  },
  mounted() {
    this.$nextTick(() => {
      this.initChart();
    });
  },
  beforeDestroy() {
    if (this.resizeTimer) {
      clearTimeout(this.resizeTimer);
      this.resizeTimer = null;
    }
  },
  methods: {
    initChart() {
      try {
        const option = this.getChartOption();

        if (this.$refs.echartRef && this.$refs.echartRef.init) {
          this.$refs.echartRef.init(option);
          this.chart = this.$refs.echartRef;
        } else {
          console.error('e-chart 组件未找到');
          return;
        }

        this.updateChart();
        this.handleResize();
      } catch (error) {
        console.error('初始化图表失败:', error);
      }
    },

    // 获取图表初始配置
    getChartOption() {
      const currentLabels = this.labels && this.labels.length >= 2 ? this.labels : ['开始', '结束'];

      return {
        backgroundColor: 'transparent',
        animation: false, // App端关闭初始动画，彻底杜绝加载抽搐
        tooltip: {
          trigger: 'axis',
          axisPointer: {
            type: 'line',
            lineStyle: { color: 'rgba(139, 92, 246, 0.3)' }
          },
          textStyle: { color: 'rgba(255, 255, 255, 0.9)', fontSize: 12 },
          backgroundColor: 'rgba(30, 30, 46, 0.8)',
          borderColor: 'rgba(139, 92, 246, 0.3)',
          borderWidth: 1,
          padding: 8,
          formatter: (params) => {
            if (!params || params.length === 0) return '';
            return params[0].name + ': ' + params[0].value.toLocaleString();
          }
        },
        grid: {
          left: 50, // 固定左边距，防止数字从无到有时图表整体左右跳动
          right: 20,
          bottom: 30,
          top: '10%',
          containLabel: false
        },
        xAxis: [{
          type: 'category',
          data: currentLabels,
          boundaryGap: false,
          axisLine: {
            show: true,
            lineStyle: { color: this.axisColor }
          },
          axisTick: { show: false },
          splitLine: { show: false },
          axisLabel: {
            color: this.textColor,
            fontSize: 10,
            interval: 0,
            formatter: (value, index) => {
              if (index === 0 || index === currentLabels.length - 1) {
                return value;
              }
              return '';
            }
          }
        }],
        yAxis: [{
          type: 'value',
          min: this.yMin,       // 【已改为动态 prop】
          max: this.yMax,       // 【已改为动态 prop】
          interval: this.yInterval, // 【已改为动态 prop】
          axisLine: {
            show: true,
            lineStyle: { color: this.axisColor }
          },
          axisTick: { show: false },
          splitLine: {
            show: true,
            lineStyle: { color: this.gridColor, type: 'dashed' }
          },
          axisLabel: {
            color: this.textColor,
            fontSize: 10,
          }
        }],
        series: [{
          name: '',
          type: 'line',
          data: this.data,
          smooth: true,
          symbol: 'circle',
          symbolSize: this.pointRadius,
          lineStyle: {
            width: 2.5,
            color: this.lineColor
          },
          itemStyle: {
            color: this.pointColor,
            borderColor: this.pointBorderColor,
            borderWidth: 2
          },
          areaStyle: {
            color: {
              type: 'linear',
              x: 0, y: 0, x2: 0, y2: 1,
              colorStops: [
                { offset: 0, color: this.gradientColors[0] },
                { offset: 1, color: this.gradientColors[1] }
              ]
            }
          }
        }]
      };
    },

    // 更新图表数据
    updateChart() {
      if (!this.chart) return;

      const currentLabels = this.labels && this.labels.length >= 2 ? this.labels : ['开始', '结束'];

      this.chart.setOption({
        xAxis: [{
          data: currentLabels,
          axisLabel: {
            interval: 0,
            formatter: (value, index) => {
              if (index === 0 || index === currentLabels.length - 1) {
                return value;
              }
              return '';
            }
          }
        }],
        // 【新加】动态更新数据时，同步将最新的 Y 轴配置塞入 setOption
        yAxis: [{
          min: this.yMin,
          max: this.yMax,
          interval: this.yInterval
        }],
        series: [{
          data: this.data
        }]
      });
    },

    // 处理窗口大小变化（防抖合并）
    handleResize() {
      if (this.chart && this.chart.resize) {
        clearTimeout(this.resizeTimer);
        this.resizeTimer = setTimeout(() => {
          this.chart.resize();
        }, 200);
      }
    }
  }
};
</script>
```
